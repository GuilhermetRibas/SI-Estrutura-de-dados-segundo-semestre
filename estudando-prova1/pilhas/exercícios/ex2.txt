typedef struct lista Lista

struc lista{
    int info;
    Lista *prox;
};

struct pilha{
    Lista *prim;
};

typedefstruct pilha Pilha;

Pilha* l_cria(){
    Pilha* p = (pilha*)malloc(Pilha);
    p->prim = NULL;
    retun p;
}

void p_push(pilha* p, int v){
    Lista* novo = (Lista*)malloc(Lista);
    novo->info = v;
    novo->prox = p->prim;
    p->prim = novo;
}

int p_pop(Pilha* p){
    if(p == NULL)exit(1);
    Lista* no = p->prim;
    int v = no->info;
    p->prim - no->prox;
    free(no);
    return v;
}

int p_tam(Pilha* p){
    int cont = 0;
    for(Lista* no = p->prim; no != NULL; no = no->prox){
        cont++;
    }
    return cont;
}

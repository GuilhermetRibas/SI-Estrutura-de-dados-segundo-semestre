-> Implementação com listas ///////////////////

struct lista {
float info;
struct lista* prox;
};

typedef struct lista Lista;

struct pilha {
Lista* prim; 
};

typedef struct pilha Pilha;

Pilha* p_cria(){
    Pilha* p = (Pilha*)malloc(sizeof(Pilha));
    p->prim = NULL;
    return p;
}

void p_push(Pilha* p, float v){
    Lista* novo = (Lista*)malloc(Lista);
    novo->info = v;
    novo->prox = p->prim;
    p->prim = novo;
}

float p_pop(Pilha* p){
    if(p == NULL)exit(1);
    Lista* no = p->prim;
    float v = no->info;
    p->prim = no->prox;
    free(no); 
    retun v; 
}

void p_libera(Pilha* p){
    Lista* no = p->prim;
    while(no != NULL){
        Lista* t = no->prox;
        free(no);
        no = t;
    }
    free(p);    
}
